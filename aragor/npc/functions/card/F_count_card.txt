function	script	cnt_card	{
.@gcid = getequipcardid;
	if (.@gcid (getarg(0),0) == getarg(1)){.@slt0=1;}
	if (.@gcid (getarg(0),1) == getarg(1)){.@slt1=1;}
	if (.@gcid (getarg(0),2) == getarg(1)){.@slt2=1;}
	if (.@gcid (getarg(0),3) == getarg(1)){.@slt3=1;}
return .@slt0+.@slt1+.@slt2+.@slt3;
}
